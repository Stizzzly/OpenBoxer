target_compile_options(ms3d_replacement PRIVATE -fno-fast-math -msse2)
set_target_properties(ms3d_replacement PROPERTIES PREFIX "" LINK_FLAGS "-Wl,--kill-at" )
target_link_options(ms3d_replacement PRIVATE -static)

add_test(NAME lighting_records_x87_and_live_callback_state COMMAND lighting_loader_tests)

add_test(NAME draw_live_sources_material_emission_and_x87 COMMAND draw_loader_tests)

target_compile_options(character_loader_tests PRIVATE -fno-fast-math -msse2)
